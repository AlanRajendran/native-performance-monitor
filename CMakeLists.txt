cmake_minimum_required(VERSION 3.24)
project(NativePerfMonitor VERSION 1.3.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with MSVC and the Windows 11 SDK on Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
add_library(perf_core STATIC src/core.cpp)
target_include_directories(perf_core PUBLIC src)
add_library(perf_native STATIC src/collector.cpp src/settings.cpp src/render.cpp src/cpu.cpp src/taskbar.cpp)
target_link_libraries(perf_native PUBLIC perf_core pdh dxgi d2d1 dwrite dwmapi user32 gdi32 shell32 advapi32 ole32 windowscodecs psapi shlwapi windowsapp uiautomationcore)
add_executable(PerfMonitor WIN32 src/app.cpp resources/app.rc)
target_link_libraries(PerfMonitor PRIVATE perf_native comctl32)
target_link_options(PerfMonitor PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest)
add_executable(CoreTests tests/core_tests.cpp)
target_link_libraries(CoreTests PRIVATE perf_core)
add_executable(NativeTests tests/native_tests.cpp)
target_link_libraries(NativeTests PRIVATE perf_native)
add_executable(WindowTests tests/window_tests.cpp)
target_link_libraries(WindowTests PRIVATE perf_native)
enable_testing()
add_test(NAME calculations COMMAND CoreTests)
add_test(NAME native_storage_and_render COMMAND NativeTests "${CMAKE_CURRENT_BINARY_DIR}/test-artifacts")
add_test(NAME native_windows COMMAND WindowTests "$<TARGET_FILE:PerfMonitor>" "${CMAKE_CURRENT_BINARY_DIR}/window-test-artifacts")
set_tests_properties(native_windows PROPERTIES RUN_SERIAL TRUE TIMEOUT 45 SKIP_RETURN_CODE 77)

add_executable(Uninstall WIN32 src/uninstall.cpp resources/uninstall.rc)
target_link_libraries(Uninstall PRIVATE user32 shell32 crypt32)
target_link_options(Uninstall PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest)

add_library(perf_install STATIC src/install.cpp)
target_link_libraries(perf_install PUBLIC perf_native)
add_executable(Setup WIN32 src/setup.cpp resources/setup.rc)
target_link_libraries(Setup PRIVATE perf_install comctl32)
target_link_options(Setup PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest)
add_executable(InstallTests tests/install_tests.cpp)
target_link_libraries(InstallTests PRIVATE perf_install)
add_test(NAME installation COMMAND InstallTests "$<TARGET_FILE:PerfMonitor>" "${CMAKE_CURRENT_BINARY_DIR}/installation-tests")
set_tests_properties(installation PROPERTIES RUN_SERIAL TRUE TIMEOUT 60)
